// sync.js — realtime sync for ASC·AR (Supabase Realtime: Broadcast + Presence) // Everyone who opens the page with the same ?room=XYZ joins the same channel. import { createClient } from 'https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2.45.4/+esm'; // Supabase dashboard → Project Settings → API // The anon key is public by design; it is safe in client code. const SUPABASE_URL = 'https://YOUR-PROJECT-REF.supabase.co'; const SUPABASE_ANON_KEY = 'YOUR-ANON-PUBLIC-KEY'; const THROTTLE_MS = 50; // max ~20 transform messages per second export function initSync({ getState, onPatch, onPeers }) { const room = new URLSearchParams(location.search).get('room') || 'default'; const clientId = crypto.randomUUID(); const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY); const channel = supabase.channel(`asc-ar-${room}`, { config: { broadcast: { self: false }, // don't echo my own messages back presence: { key: clientId }, }, }); // Someone changed something → apply it locally channel.on('broadcast', { event: 'patch' }, ({ payload }) => onPatch(payload)); // A new device joined → answer with the full current state channel.on('broadcast', { event: 'hello' }, () => { channel.send({ type: 'broadcast', event: 'patch', payload: getState() }); }); // Viewer count channel.on('presence', { event: 'sync' }, () => { onPeers(Object.keys(channel.presenceState()).length); }); channel.subscribe(async (status) => { if (status !== 'SUBSCRIBED') return; await channel.track({ joinedAt: Date.now() }); channel.send({ type: 'broadcast', event: 'hello', payload: {} }); }); // send(patch) → immediate (play, pause, step, speed) // send(patch, true) → throttled, trailing edge always delivered (rotate / scale) let pending = null; let timer = null; function flush() { timer = null; if (!pending) return; channel.send({ type: 'broadcast', event: 'patch', payload: pending }); pending = null; } function send(patch, throttle = false) { pending = { ...(pending || {}), ...patch }; if (!throttle) { clearTimeout(timer); flush(); } else if (!timer) { timer = setTimeout(flush, THROTTLE_MS); } } return { send, room }; }
